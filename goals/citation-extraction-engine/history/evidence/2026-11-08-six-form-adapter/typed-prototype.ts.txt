const range=(span:any)=>({start:span.originalStart,end:span.originalEnd});
const locate=(text:string,value:string,start:number,end:number)=>{
 const at=text.indexOf(value,start);
 if(at<start||at+value.length>end||text.indexOf(value,at+1)>=0&&text.indexOf(value,at+1)<end)throw Error('Non-unique bounded component');
 return{start:at,end:at+value.length};

const capture=(text:string,a:any)=>a?{...a,quote:text.slice(a.start,a.end)}:null;
const project=(text:string,c:any)=>{
 const span=range(c.span);let anchor=span,antecedent=null,lexicalCore=null;
 if(c.type==='supra'){

  if(matches.length!==1)throw Error('Non-unique supra token');
  const start=span.start+matches[0].index!;anchor={start,end:span.end};lexicalCore={start,end:start+5};
  antecedent=locate(text,c.partyName,span.start,start);
 }
 if(c.type==='id')lexicalCore={start:span.start,end:span.start+3};
 const pinSpan=c.spans?.pincite??c.spans?.subsection;
 const pin=pinSpan?range(pinSpan):null;const pinCite=pin?text.slice(pin.start,pin.end):null;
 const locator={pinCite,parenthetical:null};
 const dated={...locator,year:c.year===undefined?null:String(c.year),month:null,day:null};
 const casePublication={vocabularyVersion,editions:[],selectedEdition:null};
 let citation:any,form=c.type;
 if(c.type==='case'){
  form='full-case';citation={type:'case',components:{volume:String(c.volume),reporter:c.reporter,page:String(c.page)},publication:casePublication,metadata:{...dated,court:null,plaintiff:c.plaintiff??null,defendant:c.defendant??null,additionalText:null,antecedentGuess:null,shortCaseName:null,caseName:c.caseName??null}};
 }else if(c.type==='shortFormCase'){
  form='short-case';citation={type:'shortFormCase',components:{volume:String(c.volume),reporter:c.reporter},publication:casePublication,metadata:{...dated,court:null,antecedentGuess:null,plaintiff:null}};
 }else if(c.type==='id')citation={type:'id',components:{},metadata:locator};
 else if(c.type==='supra')citation={type:'supra',components:{},metadata:{...locator,antecedentGuess:text.slice(antecedent!.start,antecedent!.end),volume:c.volume===undefined?null:String(c.volume)}};
 else if(c.type==='statute'&&c.title===35||c.type==='regulation'&&c.title===37){
  form=c.type==='statute'?'usc':'cfr';citation={type:'law',components:{title:String(c.title),reporter:form==='usc'?'U.S.C.':'C.F.R.',section:c.section},publication:{editions:[],selectedEdition:null},metadata:{...dated,publisher:null}};
 }else return null;
 const fields=c.type==='case'?{volume:String(c.volume),reporter:c.reporter,page:String(c.page),pinCite}:c.type==='shortFormCase'?{volume:String(c.volume),reporter:c.reporter,pinCite}:c.type==='id'?{pinCite}:c.type==='supra'?{antecedent:c.partyName,pinCite}:{title:String(c.title),section:c.section,pinCite};
 return{form,fields,citation,anchor:capture(text,anchor),pin:capture(text,pin),antecedent:capture(text,antecedent),lexicalCore:capture(text,lexicalCore)};
};
