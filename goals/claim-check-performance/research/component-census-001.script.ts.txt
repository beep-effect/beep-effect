// Observational only: never a strict score, oracle ceiling or causal attribution.
const [replay, fixtures, out] = process.argv.slice(2);
const fs = require("node:fs");
const read = (p: string) => JSON.parse(fs.readFileSync(p, "utf8"));
const results = read(`${replay}/results.json`);
type A = { startChar: number; endChar: number; quote: string };
const ak = (a: A) => `${a.startChar}:${a.endChar}:${a.quote}`;
const sk = (a: A) => `${a.startChar}:${a.endChar}`;
const sameSet = (l: string[], r: string[]) => [...l].sort().join("\u0001") === [...r].sort().join("\u0001");
const inter = (l: string[], r: string[]) => {
  const c = new Map<string, number>();
  for (const x of r) c.set(x, (c.get(x) ?? 0) + 1);
  let n = 0;
  for (const x of l) if ((c.get(x) ?? 0) > 0) { n++; c.set(x, c.get(x)! - 1); }
  return n;
};
const bump = (m: Record<string, number>, k: string, n = 1) => { m[k] = (m[k] ?? 0) + n; };

const edgeKey = (e: any, idx: Map<string, string>, termKey = true) => {
  const n = (p: string) => idx.get(p) ?? `unmapped`;
  switch (e.kind) {
    case "part-of": return JSON.stringify([e.kind, n(e.child), n(e.parent)]);
    case "depends-on-element": return JSON.stringify([e.kind, n(e.dependent), n(e.ancestor)]);
    case "antecedent": return JSON.stringify([e.kind, n(e.referring.path), ak(e.referring.anchor), n(e.introducing.path), ak(e.introducing.anchor)]);
    case "functional": return JSON.stringify([e.kind, n(e.element), e.parts.map((p: any) => p.kind === "term" ? ["term", n(p.term.path), ak(p.term.anchor)] : ["text", ak(p.anchor)])]);
  }
  throw new Error(`edge ${e.kind}`);
};
const antecedentAnchorKey = (e: any) => JSON.stringify([ak(e.referring.anchor), ak(e.introducing.anchor)]);

const rows: any[] = [];
const agg = { spanMiss: {} as Record<string, number>, headDiff: {} as Record<string, number>, covKinds: {} as Record<string, number>, covSets: {} as Record<string, number>, findingRules: {} as Record<string, number>, kindConfusion: {} as Record<string, number>, spanSuffix: {} as Record<string, number> };

for (const fam of results.families) {
  const gold: any[] = read(`${fixtures}/${fam.familyId}/expected.json`);
  const predDoc = fs.existsSync(`${replay}/${fam.familyId}.prediction.json`) ? read(`${replay}/${fam.familyId}.prediction.json`) : null;
  const records: any[] = predDoc?.report?.records ?? [];
  const pred: any[] = records.filter((r) => r.decomposition?.status === "decomposed").map((r) => r.decomposition.tree);
  const keyOf = (claim: number, el: any) => JSON.stringify([claim, el.kind, ak(el.anchor)]);
  const gIdx = new Map<string, string>(); for (const t of gold) for (const e of t.elements) gIdx.set(e.path, keyOf(t.claimNumber, e));
  const pIdx = new Map<string, string>(); for (const t of pred) for (const e of t.elements) pIdx.set(e.path, keyOf(t.claimNumber, e));
  for (const claim of fam.claims) {
    const g = gold.find((t) => t.claimNumber === claim.claimNumber)!;
    const p = pred.find((t) => t.claimNumber === claim.claimNumber);
    const rec = records.find((r) => Number(r.decomposition?.qualification?.record?.numberText) === claim.claimNumber);
    const cov = (rec?.coverage ?? []).map((c: any) => c.kind ?? c._tag ?? "unknown");
    for (const c of new Set<string>(cov)) bump(agg.covKinds, c);
    bump(agg.covSets, [...new Set<string>(cov)].sort().join("+") || "(none)");
    for (const f of rec?.findings ?? []) bump(agg.findingRules, `${f.kind}/${f.rule}`);
    const row: any = { familyId: fam.familyId, claimNumber: claim.claimNumber, strict: claim.strict, cpc: fam.cpcSection, status: rec?.status ?? "none", coverage: [...new Set(cov)], predMissing: !p, dep: g.edges.some((e: any) => e.kind === "depends-on-element"), nested: claim.nested, functional: claim.functional, goldElements: g.elements.length };
    if (!p) { rows.push(row); continue; }
    const gS = g.elements.map((e: any) => ak(e.anchor)); const pS = p.elements.map((e: any) => ak(e.anchor));
    row.spansExact = sameSet(gS, pS);
    row.spanMatched = inter(gS, pS); row.predElements = pS.length;
    row.kindsExact = sameSet(g.elements.map((e: any) => keyOf(g.claimNumber, e)), p.elements.map((e: any) => keyOf(p.claimNumber, e)));
    // kind confusion on exact spans
    for (const ge of g.elements) { const pe = p.elements.find((e: any) => ak(e.anchor) === ak(ge.anchor)); if (pe && pe.kind !== ge.kind) bump(agg.kindConfusion, `${ge.kind}->${pe.kind}`); }
    // span miss shape
    const pOnly = p.elements.filter((e: any) => !gS.includes(ak(e.anchor)));
    for (const ge of g.elements) {
      if (pS.includes(ak(ge.anchor))) continue;
      const a = ge.anchor;
      const sameStart = pOnly.find((e: any) => e.anchor.startChar === a.startChar);
      const sameEnd = pOnly.find((e: any) => e.anchor.endChar === a.endChar);
      const overlap = pOnly.find((e: any) => e.anchor.startChar < a.endChar && a.startChar < e.anchor.endChar);
      const shape = sameStart ? (sameStart.anchor.endChar > a.endChar ? "same-start-pred-longer" : "same-start-pred-shorter") : sameEnd ? (sameEnd.anchor.startChar < a.startChar ? "same-end-pred-longer" : "same-end-pred-shorter") : overlap ? "overlap-other" : "gold-no-overlap";
      bump(agg.spanMiss, `${ge.kind}:${shape}`);
      if (sameStart && Math.abs(sameStart.anchor.endChar - a.endChar) <= 12) {
        const longer = sameStart.anchor.quote.length > a.quote.length ? sameStart.anchor.quote : a.quote;
        const shorter = sameStart.anchor.quote.length > a.quote.length ? a.quote : sameStart.anchor.quote;
        bump(agg.spanSuffix, `${sameStart.anchor.endChar > a.endChar ? "pred+" : "gold+"}${JSON.stringify(longer.slice(shorter.length))}`);
      }
    }
    // terms
    const tk = (el: any) => el.terms.map((t: any) => ak(t.anchor));
    const gT: string[] = g.elements.flatMap(tk), pT: string[] = p.elements.flatMap(tk);
    row.termAnchorsExact = sameSet(gT, pT); row.goldTerms = gT.length; row.predTerms = pT.length; row.termMatched = inter(gT, pT);
    row.termOwnershipExact = true; row.termOrderExact = true; row.termPayloadExact = true;
    for (const ge of g.elements) {
      const pe = p.elements.find((e: any) => ak(e.anchor) === ak(ge.anchor)); if (!pe) continue;
      if (!sameSet(tk(ge), tk(pe))) row.termOwnershipExact = false;
      else if (tk(ge).join("|") !== tk(pe).join("|")) row.termOrderExact = false;
    }
    const pTermBy = new Map<string, any>(); for (const e of p.elements) for (const t of e.terms) pTermBy.set(ak(t.anchor), t);
    let hd = 0, dd = 0;
    for (const e of g.elements) for (const t of e.terms) {
      const q = pTermBy.get(ak(t.anchor)); if (!q) continue;
      if (q.head !== t.head) { hd++; bump(agg.headDiff, q.head.toLowerCase().replace(/s\b/g, "") === t.head.toLowerCase().replace(/s\b/g, "") ? "plural-only" : (t.anchor.quote.toLowerCase().includes(q.head.toLowerCase()) ? "pred-head-substring-of-quote" : "other")); }
      if (q.determiner !== t.determiner) { dd++; bump(agg.headDiff, `det:${t.determiner}->${q.determiner}`); }
    }
    row.headDiffs = hd; row.detDiffs = dd; if (hd || dd) row.termPayloadExact = false;
    // edges by kind
    const kinds = ["part-of", "depends-on-element", "antecedent", "functional"];
    for (const k of kinds) {
      const ge: string[] = g.edges.filter((e: any) => e.kind === k).map((e: any) => edgeKey(e, gIdx));
      const pe: string[] = p.edges.filter((e: any) => e.kind === k).map((e: any) => edgeKey(e, pIdx));
      row[`${k}:gold`] = ge.length; row[`${k}:pred`] = pe.length; row[`${k}:matched`] = inter(ge, pe);
      row[`${k}:exact`] = sameSet(ge, pe);
    }
    // antecedent attachment independent of element ownership
    const ga = g.edges.filter((e: any) => e.kind === "antecedent").map(antecedentAnchorKey), pa = p.edges.filter((e: any) => e.kind === "antecedent").map(antecedentAnchorKey);
    row.antecedentAnchorsExact = sameSet(ga, pa);
    // defect signature
    const sig: string[] = [];
    if (!row.spansExact) sig.push("span"); else if (!row.kindsExact) sig.push("kind");
    if (!row.termAnchorsExact) sig.push("term-anchor"); else { if (!row.termOwnershipExact) sig.push("term-owner"); if (!row.termOrderExact) sig.push("term-order"); }
    if (!row.termPayloadExact) { if (hd) sig.push("head"); if (dd) sig.push("det"); }
    for (const k of kinds) if (!row[`${k}:exact`]) sig.push(`edge:${k}`);
    row.signature = sig;
    rows.push(row);
  }
}

const failed = rows.filter((r) => !r.strict);
const present = failed.filter((r) => !r.predMissing);
const count = (f: (r: any) => boolean, rs = failed) => rs.filter(f).length;
const sigs: Record<string, number> = {}; const single: Record<string, number> = {}; const anySig: Record<string, number> = {};
for (const r of present) { bump(sigs, r.signature.join("+") || "(no-diff!)"); if (r.signature.length === 1) bump(single, r.signature[0]); for (const s of r.signature) bump(anySig, s); }
const top = (m: Record<string, number>, n = 25) => Object.fromEntries(Object.entries(m).sort((a, b) => b[1] - a[1]).slice(0, n));
const nodeExact = present.filter((r) => r.spansExact && r.kindsExact);
const anchorExact = nodeExact.filter((r) => r.termAnchorsExact && r.termOwnershipExact && r.termOrderExact);

  anySignature: top(anySig), singleDefectFailures: top(single), signatures: top(sigs, 30),
  edgeTotals: Object.fromEntries(["part-of", "depends-on-element", "antecedent", "functional"].map((k) => [k, { gold: rows.reduce((s, r) => s + (r[`${k}:gold`] ?? 0), 0), pred: rows.reduce((s, r) => s + (r[`${k}:pred`] ?? 0), 0), matched: rows.reduce((s, r) => s + (r[`${k}:matched`] ?? 0), 0) }])),
  termTotals: { gold: rows.reduce((s, r) => s + (r.goldTerms ?? 0), 0), pred: rows.reduce((s, r) => s + (r.predTerms ?? 0), 0), matched: rows.reduce((s, r) => s + (r.termMatched ?? 0), 0) },
  elementTotals: { gold: rows.reduce((s, r) => s + r.goldElements, 0), pred: rows.reduce((s, r) => s + (r.predElements ?? 0), 0), spanMatched: rows.reduce((s, r) => s + (r.spanMatched ?? 0), 0) },
  bySubgroup: { dependent: { n: count((r) => r.dep, rows), strict: count((r) => r.dep && r.strict, rows) }, independent: { n: count((r) => !r.dep, rows), strict: count((r) => !r.dep && r.strict, rows) } },
  abstention: { status: top(rows.reduce((m, r) => (bump(m, r.status), m), {} as Record<string, number>)), coverageKindsAnyClaim: top(agg.covKinds), coverageSets: top(agg.covSets), findingRules: top(agg.findingRules, 20) },
  spanMiss: top(agg.spanMiss, 30), spanSuffix: top(agg.spanSuffix, 30), kindConfusion: top(agg.kindConfusion), headDiff: top(agg.headDiff),
};

"4+w" : t.anchor.quote.split(/\s+/).length + "w")); (samples.goldOnly[`${t.determiner}:${sh}`] ??= []).length < 12 && samples.goldOnly[`${t.determiner}:${sh}`].push(t.anchor.quote.slice(0, 60)); }
