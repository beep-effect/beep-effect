import * as S from "effect/Schema";
import { ImportedThing } from "./imported.ts";

export class Heading extends S.Class<Heading>("Heading")({
  level: S.Number,
}) {
    static readonly is = S.is(Heading);
    static readonly decodeUnknownSync = S.decodeUnknownSync(Heading);
}

export const isHeadingGuard = S.is(Heading);

const isImported = S.is(ImportedThing);

export const classify = (value: unknown): string => {
  if (Heading.is(value)) {
    return `heading:${Heading.decodeUnknownSync(value).level}`;
  }
  if (isImported(value)) {
    return "imported";
  }
  return "unknown";
};
