import * as S from "effect/Schema";
import { ImportedThing } from "./imported.ts";

export class Heading extends S.Class<Heading>("Heading")({
  level: S.Number,
}) {}

const isHeading = S.is(Heading);
const decodeHeading = S.decodeUnknownSync(Heading);

export const isHeadingGuard = S.is(Heading);

const isImported = S.is(ImportedThing);

export const classify = (value: unknown): string => {
  if (isHeading(value)) {
    return `heading:${decodeHeading(value).level}`;
  }
  if (isImported(value)) {
    return "imported";
  }
  return "unknown";
};

export const headingCapabilities = { isHeading, kind: "heading" };
